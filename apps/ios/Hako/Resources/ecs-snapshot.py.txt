"""Read-only, one-shot Linux snapshots invoked through ECS Cloud Assistant.

Uses an existing Python 3 and optional ss. No installation, remote network access,
scheduled task, firewall change, or persistent collector is performed.
"""
import ipaddress
import http.client
import json
import os
import platform
import pwd
import re
import shutil
import socket
import subprocess
import sys
import time
from pathlib import Path

PROC = Path('/proc')
SYSTEM_NAMES = {'systemd', 'sshd', 'cron', 'crond', 'dbus-daemon', 'dbus-broker',
                'systemd-resolve', 'systemd-network', 'systemd-journal', 'aliyun-service'}


def read(path):
    try:
        return Path(path).read_text(errors='replace')
    except (OSError, ValueError):
        return None


def number_fields(text):
    result = {}
    for line in (text or '').splitlines():
        key, _, value = line.partition(':')
        fields = value.split()
        if fields:
            try:
                result[key] = int(fields[0])
            except ValueError:
                pass
    return result


def cpu_counters(text):
    for line in (text or '').splitlines():
        fields = line.split()
        if fields and fields[0] == 'cpu':
            try:
                # guest and guest_nice are already included in user and nice.
                values = [int(value) for value in fields[1:9]]
                return values if len(values) == 8 and all(value >= 0 for value in values) else None
            except ValueError:
                return None
    return None


def cpu_usage(before, after):
    if before is None or after is None:
        return None
    deltas = [end - start for start, end in zip(before, after)]
    # iowait may decrease on Linux; clamp it without hiding resets of other counters.
    if any(value < 0 for index, value in enumerate(deltas) if index != 4):
        return None
    deltas[4] = max(0, deltas[4])
    total = sum(deltas)
    return round((total - deltas[3] - deltas[4]) / total * 100, 2) if total > 0 else None


def process_stat(text):
    # comm may contain spaces and parentheses; fields begin after the last ')'.
    if not text or ')' not in text:
        return None
    try:
        name = text[text.index('(') + 1:text.rindex(')')]
        fields = text[text.rindex(')') + 2:].split()
        return {'name': name[:120], 'parent': int(fields[1]), 'flags': int(fields[6]),
                'ticks': int(fields[11]) + int(fields[12]), 'threads': int(fields[17]),
                'start': int(fields[19]), 'rss': int(fields[21])}
    except (ValueError, IndexError):
        return None


def process_samples():
    result = {}
    for path in PROC.iterdir():
        if path.name.isdigit():
            value = process_stat(read(path / 'stat'))
            if value:
                result[int(path.name)] = value
    return result


def classify(stat, uid, cgroup, loginuid):
    if stat['flags'] & 0x200000:
        return 'system', '内核线程'
    if stat['name'] in SYSTEM_NAMES:
        return 'system', '系统基础服务（按名称推断）'
    if '/user.slice/' in cgroup or (loginuid is not None and 1000 <= loginuid < 4294967295):
        return 'user', '用户会话（按启动来源推断）'
    if 1000 <= uid < 65534:
        return 'user', '普通用户任务（按 UID 推断）'
    return 'unknown', '无法可靠确认系统或业务用途'


def processes(before, after, elapsed):
    hz = os.sysconf('SC_CLK_TCK')
    page = os.sysconf('SC_PAGE_SIZE')
    uptime = float((read(PROC / 'uptime') or '0').split()[0])
    result = []
    for pid, stat in after.items():
        path = PROC / str(pid)
        status = number_fields(read(path / 'status'))
        uid = status.get('Uid')
        if uid is None:
            continue
        try:
            user = pwd.getpwuid(uid).pw_name[:80]
        except KeyError:
            user = str(uid)
        try:
            loginuid = int((read(path / 'loginuid') or '').strip())
        except ValueError:
            loginuid = None
        category, source = classify(stat, uid, read(path / 'cgroup') or '', loginuid)
        prior = before.get(pid)
        cpu = None
        if prior and prior['start'] == stat['start'] and elapsed > 0:
            cpu = round(max(0, stat['ticks'] - prior['ticks']) / hz / elapsed * 100, 2)
        io = number_fields(read(path / 'io'))
        try:
            executable = os.readlink(path / 'exe')[:1024]
        except OSError:
            executable = None
        result.append({'pid': pid, 'parent': stat['parent'], 'name': stat['name'], 'uid': uid,
                       'user': user, 'category': category, 'source': source, 'cpu': cpu,
                       'memory': max(0, stat['rss']) * page, 'threads': stat['threads'],
                       'elapsed': max(0, uptime - stat['start'] / hz), 'executable': executable,
                       'read': io.get('read_bytes'), 'write': io.get('write_bytes')})
    return sorted(result, key=lambda p: (-(p['cpu'] or 0), -p['memory'], p['pid']))


def interfaces(text):
    result = {}
    for line in (text or '').splitlines()[2:]:
        name, _, values = line.partition(':')
        fields = values.split()
        if len(fields) >= 16:
            result[name.strip()] = {'receive': int(fields[0]), 'send': int(fields[8]),
                                    'receiveDrops': int(fields[3]), 'sendDrops': int(fields[11])}
    return result


def network(before, after, elapsed):
    result = []
    for name, values in after.items():
        item = dict(values, name=name, receiveRate=None, sendRate=None)
        if name in before and elapsed > 0:
            for key in ('receive', 'send'):
                delta = values[key] - before[name][key]
                if delta >= 0:
                    item[key + 'Rate'] = delta / elapsed
        result.append(item)
    return sorted(result, key=lambda item: (item['name'] == 'lo', item['name']))


def parse_sockets(text):
    result = []
    seen = set()
    for line in text.splitlines():
        fields = line.split(None, 6)
        if len(fields) < 6 or fields[0] not in ('tcp', 'udp'):
            continue
        match = re.search(r'\("([^"\n]+)",pid=(\d+)', fields[6] if len(fields) > 6 else '')
        item = {'protocolName': fields[0], 'state': fields[1], 'local': fields[4][:200],
                'peer': fields[5][:200], 'pid': int(match[2]) if match else None,
                'process': match[1][:120] if match else None}
        only = re.search(r'\bv6only:([01])\b', fields[6] if len(fields) > 6 else '')
        if only:
            item['v6only'] = only[1] == '1'
        key = (item['protocolName'], item['state'], item['local'], item['peer'], item['pid'])
        if key not in seen:
            seen.add(key)
            result.append(item)
    return result


def sockets(prefix=None, timeout=5):
    try:
        result = subprocess.run((prefix or []) + ['ss', '-H', '-n', '-t', '-u', '-a', '-p', '-e', '-O'],
                                capture_output=True, text=True, timeout=timeout, check=False)
        if result.returncode != 0:
            return None, '无法查询连接，请检查 ss 工具和系统权限。'
        return parse_sockets(result.stdout), None
    except (OSError, subprocess.TimeoutExpired):
        return None, '无法查询连接，请检查系统已有 ss 工具和读取权限。'


class DockerConnection(http.client.HTTPConnection):
    def __init__(self):
        super().__init__('localhost', timeout=2)

    def connect(self):
        self.sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        self.sock.settimeout(self.timeout)
        self.sock.connect('/var/run/docker.sock')


def docker_get(path):
    connection = DockerConnection()
    try:
        connection.request('GET', path)
        response = connection.getresponse()
        if response.status != 200:
            return None
        data = response.read(2_097_153)
        return json.loads(data) if len(data) <= 2_097_152 else None
    except (OSError, ValueError, http.client.HTTPException):
        return None
    finally:
        connection.close()


def docker_containers():
    # Local read-only Engine API only: never follow Docker contexts, DOCKER_HOST,
    # redirects, or TCP endpoints. No credentials or container environment read.
    version = docker_get('/version')
    if not isinstance(version, dict) or not re.fullmatch(r'1\.\d+', str(version.get('ApiVersion', ''))):
        return []
    values = docker_get('/v' + version['ApiVersion'] + '/containers/json')
    return values if isinstance(values, list) else []


def docker_ip(value):
    address = ipaddress.ip_address(value)
    return str(getattr(address, 'ipv4_mapped', None) or address)


def docker_proxy_target(pid):
    if not isinstance(pid, int) or pid <= 0:
        return None
    args = (read(PROC / str(pid) / 'cmdline') or '').rstrip('\0').split('\0')
    if not args or args[0].rsplit('/', 1)[-1] != 'docker-proxy':
        return None
    options = {}
    wanted = {'proto', 'host-ip', 'host-port', 'container-ip', 'container-port'}
    index = 1
    while index < len(args):
        key, separator, value = args[index].lstrip('-').partition('=')
        if args[index].startswith('-') and key in wanted:
            if not separator:
                index += 1
                if index >= len(args):
                    return None
                value = args[index]
            if key in options:
                return None
            options[key] = value
        index += 1
    try:
        host_port, container_port = int(options['host-port']), int(options['container-port'])
        if options['proto'] not in ('tcp', 'udp') or not all(1 <= port <= 65535 for port in (host_port, container_port)):
            return None
        return (options['proto'], docker_ip(options['host-ip']), host_port,
                docker_ip(options['container-ip']), container_port)
    except (KeyError, ValueError):
        return None


def label_docker_proxies(values, name_key):
    targets = {value.get('pid') for value in values if value.get(name_key) == 'docker-proxy'
               and value.get('namespace', 'host') == 'host'}
    if not targets:
        return
    targets = {pid: docker_proxy_target(pid) for pid in targets}
    if not any(targets.values()):
        return
    names = {}
    for container in docker_containers():
        try:
            name = container['Names'][0].lstrip('/')
            if not re.fullmatch(r'[a-zA-Z0-9][a-zA-Z0-9_.-]{0,249}', name):
                continue
            addresses = {docker_ip(network[key])
                         for network in container['NetworkSettings']['Networks'].values()
                         for key in ('IPAddress', 'GlobalIPv6Address') if network.get(key)}
            for port in container.get('Ports', []):
                if not port.get('PublicPort') or not port.get('IP'):
                    continue
                for address in addresses:
                    target = (port['Type'], docker_ip(port['IP']), int(port['PublicPort']), address, int(port['PrivatePort']))
                    names.setdefault(target, set()).add(name)
        except (KeyError, IndexError, TypeError, ValueError, AttributeError):
            continue
    resolved = {pid: 'docker:' + next(iter(names[target])) for pid, target in targets.items()
                if target in names and len(names[target]) == 1}
    # A proxy may have exited or its PID been reused during the API read.
    resolved = {pid: name for pid, name in resolved.items() if docker_proxy_target(pid) == targets[pid]}
    for value in values:
        pid = value.get('pid')
        if pid in resolved and value.get(name_key) == 'docker-proxy' and value.get('namespace', 'host') == 'host':
            value[name_key] = resolved[pid]


def network_namespaces():
    """One representative PID per namespace; host-network containers share self."""
    candidates = {}
    incomplete = False
    try:
        own = os.readlink(PROC / 'self' / 'ns' / 'net')
        for path in PROC.iterdir():
            if not path.name.isdigit():
                continue
            try:
                name = os.readlink(path / 'ns' / 'net')
                if name != own:
                    candidates.setdefault(name, []).append(int(path.name))
            except FileNotFoundError:
                pass  # Processes can exit while taking a snapshot.
            except OSError:
                incomplete = True
    except OSError:
        return [], True
    return [(name, sorted(pids)) for name, pids in sorted(candidates.items())], incomplete


def all_sockets():
    values, error = sockets()
    if values is None:
        return values, error
    for value in values:
        value['namespace'] = 'host'
    namespaces, incomplete = network_namespaces()
    nsenter = shutil.which('nsenter')
    deadline = time.monotonic() + 8
    if len(namespaces) > 32 or (namespaces and not nsenter):
        incomplete = True
    for name, pids in namespaces[:32] if nsenter else []:
        rows = None
        for pid in pids[:2]:
            remaining = deadline - time.monotonic()
            if remaining <= 0:
                break
            # Pin the target namespace through its /proc file; never enter its
            # mount namespace or run a container-provided executable.
            rows, issue = sockets([nsenter, '--net=' + str(PROC / str(pid) / 'ns' / 'net'), '--'],
                                  timeout=min(2, remaining))
            if rows is not None:
                break
        if rows is None:
            incomplete = True
            continue
        for value in rows:
            value['namespace'] = name
        values.extend(rows)
    warning = '部分网络命名空间无法读取，连接数可能偏少；请检查 Linux 用户权限与 nsenter 工具。' if incomplete else None
    return values, warning


def socket_endpoint(value):
    try:
        host, port = value.rsplit(':', 1)
        port = int(port)
        if not 1 <= port <= 65535:
            return None
        host = host.strip('[]')
        if host == '*':
            return host, port, 0
        address = ipaddress.ip_address(host.split('%', 1)[0])
        # Preserve the socket family for matching an IPv6 wildcard listener,
        # while canonicalizing mapped addresses for exact-address comparison.
        mapped = getattr(address, 'ipv4_mapped', None)
        normalized = str(mapped or address)
        if address.version == 6 and not mapped and '%' in host:
            normalized += '%' + host.split('%', 1)[1]
        return normalized, port, address.version
    except (ValueError, AttributeError):
        return None


def is_listener(value):
    return value['state'] == 'LISTEN' or (value['protocolName'] == 'udp' and value['state'] == 'UNCONN')


def connection_summary(values):
    # This is a snapshot inference, not a record of the original TCP handshake.
    # Only established TCP sockets are counted; UDP has no accepted sessions.
    listeners = [(socket_endpoint(v['local']), v) for v in values
                 if v['protocolName'] == 'tcp' and v['state'] == 'LISTEN']
    listeners = [(endpoint, v) for endpoint, v in listeners if endpoint]
    counts = dict(outgoing=0, incoming=0, unknown=0)
    for value in values:
        if value['protocolName'] != 'tcp' or value['state'] not in ('ESTAB', 'ESTABLISHED'):
            continue
        local = socket_endpoint(value['local'])
        direction = 'unknown'
        if local:
            host, port, family = local
            accepted = any(listener.get('namespace') == value.get('namespace') and p == port and
                (h == host or (h == '*' and (not listener.get('v6only') or family == 6)) or
                 (f == family and h in ('0.0.0.0', '::')) or
                 (h == '::' and family == 4 and listener.get('v6only') is False))
                for (h, p, f), listener in listeners)
            direction = 'incoming' if accepted else 'outgoing'
        value['direction'] = direction
        counts[direction] += 1
    return counts


def disks():
    candidates = []
    for line in (read(PROC / 'self' / 'mountinfo') or '').splitlines():
        fields = line.split()
        try:
            split = fields.index('-')
            filesystem = fields[split + 1]
            mount = re.sub(r'\\([0-7]{3})', lambda m: chr(int(m[1], 8)), fields[4])
            # Each container overlay has its own synthetic device ID but reports
            # the backing disk's capacity. Only keep overlay if it is our root FS.
            if filesystem not in ('ext2', 'ext3', 'ext4', 'xfs', 'btrfs', 'zfs', 'vfat', 'f2fs', 'nfs', 'nfs4') and not (filesystem == 'overlay' and mount == '/'):
                continue
            candidates.append((fields[2], fields[3], mount, filesystem))
        except (ValueError, IndexError):
            continue
    # Prefer / and filesystem roots over bind mounts, regardless of mount order.
    candidates.sort(key=lambda item: (item[2] != '/', item[1] != '/', item[2].count('/'), item[2]))
    result, seen = [], set()
    for device, root, mount, filesystem in candidates:
        if device in seen:
            continue
        try:
            stat = os.statvfs(mount)
            if stat.f_blocks <= 0:
                continue
            seen.add(device)
            result.append({'mount': mount[:512], 'filesystem': filesystem,
                           'total': stat.f_blocks * stat.f_frsize,
                           'used': (stat.f_blocks - stat.f_bfree) * stat.f_frsize,
                           'available': stat.f_bavail * stat.f_frsize})
        except OSError:
            continue
    return result


def bounded_output(result, limit=20_000):
    # Cloud Assistant truncates output at 24 KiB; keep UTF-8 comfortably below it.
    result['truncated'] = False
    while True:
        text = json.dumps(result, ensure_ascii=True, separators=(',', ':'), allow_nan=False)
        if len(text.encode()) <= limit:
            return text
        candidates = [key for key in ('processes', 'sockets', 'disks', 'interfaces') if result.get(key)]
        if not candidates:
            raise ValueError('snapshot too large')
        key = max(candidates, key=lambda k: len(json.dumps(result[k])))
        result[key].pop()
        result['truncated'] = True


def collect(mode, pid=0):
    result = {'version': 1, 'time': time.time()}
    if mode in ('processes', 'process'):
        first = process_samples()
        start = time.monotonic()
        time.sleep(1)
        second = process_samples()
        values = processes(first, second, time.monotonic() - start)
        result['totalProcesses'] = len(values)
        result['processes'] = [p for p in values if mode != 'process' or p['pid'] == pid]
        label_docker_proxies(result['processes'], 'name')
        result['memoryTotal'] = number_fields(read(PROC / 'meminfo')).get('MemTotal', 0) * 1024
    elif mode == 'network':
        first = interfaces(read(PROC / 'net' / 'dev'))
        start = time.monotonic()
        time.sleep(1)
        second = interfaces(read(PROC / 'net' / 'dev'))
        result['interfaces'] = [v for v in network(first, second, time.monotonic() - start) if v['name'] == 'eth0']
        result['primaryInterface'] = 'eth0' if result['interfaces'] else None
        result['sockets'], result['socketError'] = all_sockets()
        result['totalSockets'] = len(result['sockets'] or [])
        if result['sockets'] is not None:
            # Container listeners are needed to classify their accepted sockets,
            # but are not host listeners and must not appear in the host port list.
            result['socketSummary'] = connection_summary(result['sockets'])
            result['sockets'] = [v for v in result['sockets']
                                 if not is_listener(v) or v.get('namespace', 'host') == 'host']
            label_docker_proxies(result['sockets'], 'process')
            result['totalSockets'] = len(result['sockets'])
            result['totalListeningPorts'] = len({(v['protocolName'], socket_endpoint(v['local'])[1])
                for v in result['sockets'] if is_listener(v) and socket_endpoint(v['local'])})
            result['sockets'].sort(key=lambda v: (not is_listener(v), v['protocolName'], v['local'], v['peer']))
    elif mode == 'overview':
        first = cpu_counters(read(PROC / 'stat'))
        time.sleep(1)
        result['cpuUsagePercent'] = cpu_usage(first, cpu_counters(read(PROC / 'stat')))
        mem = number_fields(read(PROC / 'meminfo'))
        result.update(hostname=platform.node()[:200], kernel=platform.release()[:200],
                      architecture=platform.machine()[:80], disks=disks())
        if result['disks']:
            result['diskTotal'] = sum(disk['total'] for disk in result['disks'])
            result['diskUsed'] = sum(disk['used'] for disk in result['disks'])
        result['memoryTotal'] = mem.get('MemTotal', 0) * 1024
        if 'MemAvailable' in mem:
            result['memoryAvailable'] = mem['MemAvailable'] * 1024
        result['uptime'] = float((read(PROC / 'uptime') or '0').split()[0])
        result['load'] = [float(v) for v in (read(PROC / 'loadavg') or '').split()[:3]]
    else:
        raise ValueError('unknown mode')
    result['time'] = time.time()
    return bounded_output(result)


if __name__ == '__main__':
    print(collect(sys.argv[1], int(sys.argv[2]) if len(sys.argv) > 2 else 0))
